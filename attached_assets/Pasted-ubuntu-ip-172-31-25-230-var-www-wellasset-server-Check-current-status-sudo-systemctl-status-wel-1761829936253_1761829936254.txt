# The service is trying to use start.sh which doesn't exist yet
# Let's manually create it on the server

# Create the start.sh script
sudo tee /var/www/wellasset/start.sh > /dev/null << 'EOF'
#!/bin/bash
# Well Asset Real Estate Platform - Startup Wrapper Script
set -e

# Load environment variables
set -a
source /var/www/wellasset/.env.production
set +a

# Start PM2 in runtime mode (foreground)
exec /home/nodejs/node_modules/.bin/pm2-runtime start /var/www/wellasset/ecosystem.config.cjs --env production
EOF

# Make it executable
sudo chmod +x /var/www/wellasset/start.sh
sudo chown nodejs:nodejs /var/www/wellasset/start.sh

# Restart the service
sudo systemctl restart wellasset

# Check status
sleep 5
sudo systemctl status wellasset

# Test health
curl http://localhost:5000/health
